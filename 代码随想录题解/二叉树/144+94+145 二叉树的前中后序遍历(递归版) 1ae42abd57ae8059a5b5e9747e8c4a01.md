```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<Integer> preorderTraversal(TreeNode root) {
        List<Integer> ansList = new ArrayList<>();
        preOrder(root, ansList);
        //inOrder(root, ansList);
        //postOrder(root, ansList);
        return ansList;
    }

    public void preOrder(TreeNode root, List<Integer> ansList){
        if(root == null){
            return;
        }else{
            ansList.add(root.val);
            preOrder(root.left, ansList);
            preOrder(root.right, ansList);
        }
    }
    
    public void inOrder(TreeNode root, List<Integer> ansList){
        if(root == null){
            return;
        }else{
            inOrder(root.left, ansList);
            ansList.add(root.val);
            inOrder(root.right, ansList);
        }
    }
    
    public void postOrder(TreeNode root, List<Integer> ansList){
        if(root == null){
            return;
        }else{
            postOrder(root.left, ansList);
            postOrder(root.right, ansList);
            ansList.add(root.val);
        }
    }
}
}
```
