# 198. 打家劫舍

[https://leetcode.cn/problems/house-robber/description/](https://leetcode.cn/problems/house-robber/description/)

比较简单。先定义dp数组的含义。由题可直接将dp[i]定义为当只有下标为0~i的房屋可以被偷时，能够偷窃到的最高金额为dp[i]。接下来确定dp数组的递推公式，对于一间房屋，只有两种选择：要么偷，要么不偷。如果选择偷的话，要注意相邻的房屋不能一起偷，此时dp[i] = dp[i - 2] + nums[i]；如果选择不偷的话，那么dp[i] = dp[i - 1]。最后比较这两个选择，看看哪个选择可以使得偷窃的金额最大，dp[i]的值就为这个金额最大值。代码如下：

```java
// 常规版（一维dp）
class Solution {
    public int rob(int[] nums) {
        int[] dp = new int[nums.length];
        if(nums.length == 1){
            return nums[0];
        }
        if(nums.length == 2){
            return Math.max(nums[0], nums[1]);
        }

        dp[0] = nums[0];
        dp[1] = Math.max(nums[0], nums[1]);
        for(int i = 2; i < nums.length; i++){
            dp[i] = Math.max(dp[i - 2] + nums[i], dp[i - 1]);
        }
        return dp[nums.length - 1];
    }
}

// 空间优化版（滚动变量）
class Solution {
    public int rob(int[] nums) {
        if(nums == null || nums.length == 0){
            return 0;
        }

        int prev2 = 0; // 相当于 dp[i-2]
        int prev1 = 0; // 相当于 dp[i-1]
        int res = 0;
        for(int num : nums){
            res = Math.max(prev2 + num, prev1);
            
            // 状态滚动更新
            prev2 = prev1;
            prev1 = res;
        }
        return res;
    }
}

// 补充二维dp解法
class Solution {
    public int rob(int[] nums) {
        if (nums == null || nums.length == 0) return 0;
        int n = nums.length;
        
        // dp[i][0] 表示不抢第 i 间房能拿到的最大钱
        // dp[i][1] 表示抢第 i 间房能拿到的最大钱
        int[][] dp = new int[n][2];
        
        // 初始化第一间房
        dp[0][0] = 0;           // 不抢第一间
        dp[0][1] = nums[0];     // 抢第一间
        
        for (int i = 1; i < n; i++) {
            // 今天不抢：那么昨天可以抢，也可以不抢，取最大值
            dp[i][0] = Math.max(dp[i-1][0], dp[i-1][1]);
            
            // 今天要抢：那么昨天【绝对不能】抢
            dp[i][1] = dp[i-1][0] + nums[i];
        }
        
        // 最后一间房抢或不抢，取最大值
        return Math.max(dp[n - 1][0], dp[n - 1][1]);
    }
}
```
